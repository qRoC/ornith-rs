def rolling_median(xs, k):
    """Median over a sliding window, kept sorted by insertion."""
    import bisect
    window, out = [], []
    for i, x in enumerate(xs):
        bisect.insort(window, x)
        if len(window) > k:
            window.pop(bisect.bisect_left(window, xs[i - k]))
        if len(window) == k:
            out.append(window[k // 2] if k % 2 else
                       (window[k // 2 - 1] + window[k // 2]) / 2)
    return out

pub fn merge_spans(mut spans: Vec<(u32, u32)>) -> Vec<(u32, u32)> {
    spans.sort_unstable();
    let mut out: Vec<(u32, u32)> = Vec::with_capacity(spans.len());
    for (lo, hi) in spans {
        match out.last_mut() {
            Some(last) if lo <= last.1 => last.1 = last.1.max(hi),
            _ => out.push((lo, hi)),
        }
    }
    out
}

A gear train multiplies torque by the ratio of tooth counts, and the ratio is
exact because teeth are counted, not measured. That is why clocks were built
from gears rather than from levers: an error in a lever accumulates with every
stroke, while a gear that has meshed a thousand times has advanced exactly a
thousand teeth. The escapement is the one place where the count is turned back
into time, and it is the one place where a clock's accuracy is decided.
